## Usage

{/* prettier-ignore */}
```tsx
import { Frame } from "@elmeragroup/fuse/frame";

<Frame.Root>
  <Frame.Header>
    <Frame.Title>Usage</Frame.Title>
  </Frame.Header>
  <Frame.Panel>March, 1 240 kWh</Frame.Panel>
  <Frame.Panel>April, 980 kWh</Frame.Panel>
</Frame.Root>
```
