## Usage

{/* prettier-ignore */}
```tsx
import { Combobox } from "@elmeragroup/fuse/combobox";

<Combobox.Root items={["Apple", "Pear"]}>
  <Combobox.Input aria-label="Fruit" placeholder="Search fruit…" />
  <Combobox.Content>
    <Combobox.Empty />
    <Combobox.List>
      <Combobox.Collection>
        {(item: string) => (
          <Combobox.Item key={item} value={item}>
            {item}
          </Combobox.Item>
        )}
      </Combobox.Collection>
    </Combobox.List>
  </Combobox.Content>
</Combobox.Root>
```
