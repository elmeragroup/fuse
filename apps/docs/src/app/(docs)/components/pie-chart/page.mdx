## Usage

{/* prettier-ignore */}
```tsx
import { PieChart } from "@elmeragroup/fuse/chart";
import "@elmeragroup/fuse/chart.css";

<PieChart.Root data={data} config={config} nameKey="part" valueKey="kr" aria-label="Electricity bill">
  <PieChart.Pie />
  <PieChart.Labels />
  <PieChart.Tooltip />
  <PieChart.Table />
</PieChart.Root>
```
