## Usage

{/* prettier-ignore */}
```tsx
import { Sheet } from "@elmeragroup/fuse/sheet";

<Sheet.Root>
  <Sheet.Trigger>Open meter details</Sheet.Trigger>
  <Sheet.Content>
    <Sheet.Header>
      <Sheet.Title>Meter details</Sheet.Title>
    </Sheet.Header>
    <Sheet.Body>Last reading 12 480 kWh.</Sheet.Body>
  </Sheet.Content>
</Sheet.Root>
```
