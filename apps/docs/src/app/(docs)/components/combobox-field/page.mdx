## Usage

{/* prettier-ignore */}
```tsx
import { Combobox } from "@elmeragroup/fuse/combobox";
import { ComboboxField } from "@elmeragroup/fuse/combobox-field";

<ComboboxField label="Country" items={["Norway", "Sweden"]} placeholder="Search countries…">
  <Combobox.Empty />
  <Combobox.List>
    {(country: string) => (
      <Combobox.Item key={country} value={country}>
        {country}
      </Combobox.Item>
    )}
  </Combobox.List>
</ComboboxField>
```
