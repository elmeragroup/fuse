## Usage

{/* prettier-ignore */}
```tsx
import { Radio, RadioGroup } from "@elmeragroup/fuse/radio-group";

<RadioGroup label="Contract" defaultValue="fixed">
  <Radio value="fixed">Fixed price</Radio>
  <Radio value="spot">Spot price</Radio>
</RadioGroup>
```
