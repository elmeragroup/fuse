## Usage

{/* prettier-ignore */}
```tsx
import { Checkbox } from "@elmeragroup/fuse/checkbox";
import { Field } from "@elmeragroup/fuse/field";

<Field.Root orientation="horizontal">
  <Checkbox />
  <Field.Label>Accept terms</Field.Label>
</Field.Root>
```
