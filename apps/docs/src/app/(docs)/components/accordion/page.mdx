## Usage

{/* prettier-ignore */}
```tsx
import { Accordion } from "@elmeragroup/fuse/accordion";

<Accordion.Root>
  <Accordion.Item value="shipping">
    <Accordion.Header>
      <Accordion.Trigger>Shipping</Accordion.Trigger>
    </Accordion.Header>
    <Accordion.Content>Delivered within 3 to 5 business days.</Accordion.Content>
  </Accordion.Item>
</Accordion.Root>
```
