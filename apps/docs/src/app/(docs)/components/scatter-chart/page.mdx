## Usage

{/* prettier-ignore */}
```tsx
import { ScatterChart } from "@elmeragroup/fuse/chart";
import "@elmeragroup/fuse/chart.css";

<ScatterChart.Root
  data={data}
  config={config}
  xKey="tempC"
  yKey="kwh"
  seriesKey="series"
  aria-label="Weekly use against mean temperature">
  <ScatterChart.XAxis />
  <ScatterChart.YAxis />
  <ScatterChart.Points dataKey="house" />
  <ScatterChart.Tooltip />
  <ScatterChart.Table />
</ScatterChart.Root>
```
