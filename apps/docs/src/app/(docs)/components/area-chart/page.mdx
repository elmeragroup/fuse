## Usage

{/* prettier-ignore */}
```tsx
import { AreaChart } from "@elmeragroup/fuse/chart";
import "@elmeragroup/fuse/chart.css";

<AreaChart.Root data={data} config={config} xKey="month" xScale="point" aria-label="Consumption per month">
  <AreaChart.YAxis />
  <AreaChart.XAxis />
  <AreaChart.Area dataKey="kwh" />
  <AreaChart.Line dataKey="kwh" />
  <AreaChart.Tooltip />
  <AreaChart.Table />
</AreaChart.Root>
```
