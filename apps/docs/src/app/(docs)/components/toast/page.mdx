## Usage

{/* prettier-ignore */}
```tsx
import { Toast } from "@elmeragroup/fuse/toast";

<Toast.Provider>
  <SaveButton />
  <Toast.Viewport />
</Toast.Provider>

function SaveButton() {
  const toastManager = Toast.useToastManager();
  return <button onClick={() => toastManager.add({ title: "Draft saved" })}>Save draft</button>;
}
```
