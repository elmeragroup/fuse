## Usage

{/* prettier-ignore */}
```tsx
import { GaugeChart } from "@elmeragroup/fuse/chart";
import "@elmeragroup/fuse/chart.css";

<GaugeChart.Root value={3.9} max={5} valueFormat={energy} aria-label="This hour's consumption">
  <GaugeChart.Track />
  <GaugeChart.Fill />
  <GaugeChart.Center>
    <GaugeChart.CenterValue />
    <GaugeChart.CenterLabel>This hour</GaugeChart.CenterLabel>
  </GaugeChart.Center>
</GaugeChart.Root>
```
