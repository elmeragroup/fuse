## Usage

{/* prettier-ignore */}
```tsx
import { RadarChart } from "@elmeragroup/fuse/chart";
import "@elmeragroup/fuse/chart.css";

<RadarChart.Root data={data} config={config} nameKey="day" aria-label="Mean day per weekday">
  <RadarChart.Grid />
  <RadarChart.AngleAxis />
  <RadarChart.RadiusAxis />
  <RadarChart.Area dataKey="kwh" />
  <RadarChart.Line dataKey="kwh" />
  <RadarChart.Tooltip />
  <RadarChart.Table />
</RadarChart.Root>
```
