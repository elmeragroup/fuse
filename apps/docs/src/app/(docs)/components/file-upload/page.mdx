## Usage

{/* prettier-ignore */}
```tsx
import { Button } from "@elmeragroup/fuse/button";
import { FileUpload } from "@elmeragroup/fuse/file-upload";

<FileUpload.Root accept="image/*,.pdf" multiple onSelect={setFiles}>
  <FileUpload.Dropzone>
    <FileUpload.Trigger render={<Button size="sm" />}>Add files</FileUpload.Trigger>
    <FileUpload.Description>Images or PDFs.</FileUpload.Description>
  </FileUpload.Dropzone>
</FileUpload.Root>
```
