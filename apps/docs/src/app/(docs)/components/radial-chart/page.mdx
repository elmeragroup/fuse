## Usage

{/* prettier-ignore */}
```tsx
import { RadialChart } from "@elmeragroup/fuse/chart";
import "@elmeragroup/fuse/chart.css";

<RadialChart.Root data={data} config={config} nameKey="budget" valueKey="kwh" maxKey="limit" aria-label="Usage against budget">
  <RadialChart.Track />
  <RadialChart.Bar />
  <RadialChart.Tooltip />
  <RadialChart.Table />
</RadialChart.Root>
```
