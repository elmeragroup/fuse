## Usage

{/* prettier-ignore */}
```tsx
import { Card } from "@elmeragroup/fuse/card";

<Card.Root>
  <Card.Header>
    <Card.Title>March usage</Card.Title>
    <Card.Description>Estimated consumption for the period.</Card.Description>
  </Card.Header>
  <Card.Content>1 240 kWh across two meters.</Card.Content>
</Card.Root>
```
