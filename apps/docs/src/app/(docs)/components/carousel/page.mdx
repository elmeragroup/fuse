## Usage

{/* prettier-ignore */}
```tsx
import { Carousel } from "@elmeragroup/fuse/carousel";

<Carousel.Root aria-label="Electricity offers">
  <Carousel.Viewport>
    <Carousel.Slide>Spot price</Carousel.Slide>
    <Carousel.Slide>Fixed price</Carousel.Slide>
  </Carousel.Viewport>
  <Carousel.Controls>
    <Carousel.Previous />
    <Carousel.Indicators />
    <Carousel.Next />
  </Carousel.Controls>
</Carousel.Root>
```
