## Usage

{/* prettier-ignore */}
```tsx
import { MagnifyingGlass } from "@elmeragroup/fuse/icons";
import { InputGroup } from "@elmeragroup/fuse/input-group";

<InputGroup.Root>
  <InputGroup.Addon>
    <MagnifyingGlass />
  </InputGroup.Addon>
  <InputGroup.Input aria-label="Search customers" />
</InputGroup.Root>
```
