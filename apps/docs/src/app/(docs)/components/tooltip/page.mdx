## Usage

{/* prettier-ignore */}
```tsx
import { Tooltip } from "@elmeragroup/fuse/tooltip";

<Tooltip.Root>
  <Tooltip.Trigger>Grid rent</Tooltip.Trigger>
  <Tooltip.Content>The fee for using the electricity grid</Tooltip.Content>
</Tooltip.Root>
```
