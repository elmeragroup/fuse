## Usage

{/* prettier-ignore */}
```tsx
import { Autocomplete } from "@elmeragroup/fuse/autocomplete";
import { AutocompleteField } from "@elmeragroup/fuse/autocomplete-field";

<AutocompleteField label="Municipality" items={["Bergen", "Oslo"]} placeholder="Type a municipality…">
  <Autocomplete.Empty />
  <Autocomplete.List>
    {(municipality: string) => (
      <Autocomplete.Item key={municipality} value={municipality}>
        {municipality}
      </Autocomplete.Item>
    )}
  </Autocomplete.List>
</AutocompleteField>
```
