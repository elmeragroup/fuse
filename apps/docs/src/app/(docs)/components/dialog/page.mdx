## Usage

{/* prettier-ignore */}
```tsx
import { Dialog } from "@elmeragroup/fuse/dialog";

<Dialog.Root>
  <Dialog.Trigger>Move the meter reading</Dialog.Trigger>
  <Dialog.Content>
    <Dialog.Header>
      <Dialog.Title>Move the meter reading</Dialog.Title>
      <Dialog.Description>The old contract is closed.</Dialog.Description>
    </Dialog.Header>
  </Dialog.Content>
</Dialog.Root>
```
