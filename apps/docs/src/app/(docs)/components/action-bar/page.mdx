## Usage

{/* prettier-ignore */}
```tsx
import { ActionBar } from "@elmeragroup/fuse/action-bar";

<div className="relative">
  <ActionBar.Root count={selected.length}>
    <ActionBar.Count />
    <ActionBar.Clear onClick={() => setSelected([])} />
    <ActionBar.Separator />
    <ActionBar.Button>Export</ActionBar.Button>
  </ActionBar.Root>
</div>
```
