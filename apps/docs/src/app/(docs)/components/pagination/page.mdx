## Usage

{/* prettier-ignore */}
```tsx
import { Pagination } from "@elmeragroup/fuse/pagination";

<Pagination.Root>
  <Pagination.Content>
    <Pagination.Item>
      <Pagination.Previous href="?page=1" />
    </Pagination.Item>
    <Pagination.Item>
      <Pagination.Link href="?page=2" isActive>
        2
      </Pagination.Link>
    </Pagination.Item>
    <Pagination.Item>
      <Pagination.Next href="?page=3" />
    </Pagination.Item>
  </Pagination.Content>
</Pagination.Root>
```
