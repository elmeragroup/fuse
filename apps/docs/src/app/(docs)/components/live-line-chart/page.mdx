## Usage

{/* prettier-ignore */}
```tsx
import { LiveLineChart } from "@elmeragroup/fuse/chart";
import "@elmeragroup/fuse/chart.css";

<LiveLineChart.Root data={readings} label="Power" timeZone="Europe/Oslo" aria-label="Power draw, live">
  <LiveLineChart.YAxis />
  <LiveLineChart.XAxis />
  <LiveLineChart.Line />
  <LiveLineChart.LiveDot />
  <LiveLineChart.Tooltip />
  <LiveLineChart.Table />
</LiveLineChart.Root>
```
