## Usage

{/* prettier-ignore */}
```tsx
import { BarChart } from "@elmeragroup/fuse/chart";
import "@elmeragroup/fuse/chart.css";

<BarChart.Root data={data} config={config} xKey="month" aria-label="Consumption per month">
  <BarChart.YAxis />
  <BarChart.XAxis />
  <BarChart.Bar dataKey="kwh" />
  <BarChart.Tooltip />
  <BarChart.Table />
</BarChart.Root>
```
