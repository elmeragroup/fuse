## Usage

{/* prettier-ignore */}
```tsx
import { Sidebar } from "@elmeragroup/fuse/sidebar";

<Sidebar.Provider>
  <Sidebar.Root>
    <Sidebar.Content>
      <Sidebar.Group>
        <Sidebar.GroupLabel>Sales</Sidebar.GroupLabel>
        <Sidebar.GroupContent>
          <Sidebar.Menu>
            <Sidebar.MenuItem>
              <Sidebar.MenuButton isActive>Orders</Sidebar.MenuButton>
            </Sidebar.MenuItem>
          </Sidebar.Menu>
        </Sidebar.GroupContent>
      </Sidebar.Group>
    </Sidebar.Content>
    <Sidebar.Rail />
  </Sidebar.Root>
  <Sidebar.Inset>
    <Sidebar.Trigger />
  </Sidebar.Inset>
</Sidebar.Provider>
```
