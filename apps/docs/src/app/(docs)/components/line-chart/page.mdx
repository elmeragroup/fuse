## Usage

{/* prettier-ignore */}
```tsx
import { LineChart } from "@elmeragroup/fuse/chart";
import "@elmeragroup/fuse/chart.css";

<LineChart.Root data={data} config={config} xKey="month" aria-label="Consumption per month">
  <LineChart.YAxis />
  <LineChart.XAxis />
  <LineChart.Line dataKey="kwh" />
  <LineChart.Tooltip />
  <LineChart.Table />
</LineChart.Root>
```
