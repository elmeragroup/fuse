## Usage

{/* prettier-ignore */}
```tsx
import { ClockChart } from "@elmeragroup/fuse/chart";
import "@elmeragroup/fuse/chart.css";

<ClockChart.Root data={hours} config={config} xKey="start" cycle="day" interval="hour" timeZone="Europe/Oslo" aria-label="Consumption per hour">
  <ClockChart.Grid />
  <ClockChart.AngleAxis />
  <ClockChart.RadiusAxis />
  <ClockChart.Column dataKey="kwh" />
  <ClockChart.Tooltip />
  <ClockChart.Table />
</ClockChart.Root>
```
