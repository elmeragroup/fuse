## Usage

{/* prettier-ignore */}
```tsx
import { Table } from "@elmeragroup/fuse/table";

<Table.Root>
  <Table.Header>
    <Table.Row>
      <Table.Head>Order</Table.Head>
    </Table.Row>
  </Table.Header>
  <Table.Body>
    <Table.Row>
      <Table.Cell>#1042</Table.Cell>
    </Table.Row>
  </Table.Body>
</Table.Root>
```
