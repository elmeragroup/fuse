## Usage

{/* prettier-ignore */}
```tsx
import { FileTrigger } from "@elmeragroup/fuse/file-trigger";

<FileTrigger accept="image/png,.pdf" onSelect={setFiles}>
  Attach file
</FileTrigger>
```
